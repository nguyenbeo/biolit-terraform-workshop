# https://registry.terraform.io/providers/hashicorp/kubernetes/latest/docs/resources/deployment_v1
# A Pod is the smallest runnable unit in Kubernetes
# A Deployment manages Pods

resource "kubernetes_deployment_v1" "nginx" {
  metadata {
    name      = "nginx-demo"
    namespace = kubernetes_namespace_v1.name.metadata[0].name
    labels = {
      app = "nginx"
    }
  }

  spec {
    replicas = 2

    selector {
      match_labels = {
        app = "nginx"
      }
    }

    template {
      metadata {
        labels = {
          app = "nginx"
        }
      }

      spec {
        container {
          name  = "nginx"
          image = "nginx:1.27.1-alpine"

          port {
            container_port = 80
            name           = "http"
          }

          resources {
            limits = {
              cpu    = "250m"
              memory = "128Mi"
            }
            requests = {
              cpu    = "100m"
              memory = "64Mi"
            }
          }
        }

        restart_policy = "Always"
      }
    }
  }
}
